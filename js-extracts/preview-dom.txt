        { id: 'f', x: 520, y: 200, label: 'F' },
        { id: 'g', x: 710, y: 80, label: 'G' },
        { id: 'h', x: 710, y: 170, label: 'H' }
      ],
      edges: [
        { from: 'a', to: 'b', w: 4 }, { from: 'a', to: 'c', w: 2 },
        { from: 'b', to: 'd', w: 5 }, { from: 'b', to: 'e', w: 3 },
        { from: 'c', to: 'e', w: 2 }, { from: 'c', to: 'f', w: 6 },
        { from: 'd', to: 'g', w: 4 }, { from: 'e', to: 'g', w: 2 },
        { from: 'e', to: 'h', w: 5 }, { from: 'f', to: 'h', w: 3 }
      ]
    };
  }

  function presetFor(id) {
    if (id === 'reverse-proxy') return proxyPreset();
    if (id === 'chart') return chartPreset('bar');
    if (id === 'pipeline') return pipelinePreset();
    if (id === 'graph') return graphPreset();
    return flowPreset();
  }

  var state = { spec: flowPreset() };
  var stage = document.getElementById('stage');
  var specBox = document.getElementById('spec');

  function fillSelect(sel, items) {
    items.forEach(function (it) {
      var o = document.createElement('option');
      o.value = it.id;
      o.textContent = it.label;
      sel.appendChild(o);
    });
  }

  fillSelect(document.getElementById('cable'), CABLES);
  fillSelect(document.getElementById('motion'), MOTIONS);

  function syncControls() {
    var s = state.spec;
    document.getElementById('module').value = s.module || 'flow';
    document.getElementById('chartTypeRow').style.display = s.module === 'chart' ? '' : 'none';
    if (s.module === 'chart') document.getElementById('chartType').value = s.type || 'bar';
    document.getElementById('cable').value = s.cable || 'solid';
    document.getElementById('motion').value = s.motion || 'flow';
    document.getElementById('speed').value = s.speed || 1;
    document.getElementById('speedVal').textContent = (s.speed || 1).toFixed(2).replace(/0$/, '') + '×';
    specBox.value = JSON.stringify(s, null, 2);
  }

  function render() {
    if (stage.__alur) {
      Alur.destroy(stage.__alur);
      stage.__alur = null;
    }
    stage.innerHTML = '';
    stage.setAttribute('data-module', state.spec.module || 'flow');
    stage.setAttribute('data-spec', JSON.stringify(state.spec));
    stage.removeAttribute('data-ready');
    Alur.mount(stage);
  }

  function setAllEdges(patch) {
    (state.spec.edges || []).forEach(function (e) {
      Object.keys(patch).forEach(function (k) { e[k] = patch[k]; });
    });
  }

  document.getElementById('render').addEventListener('click', function () {
    render();
  });

  document.getElementById('module').addEventListener('change', function (ev) {
    state.spec = presetFor(ev.target.value);
    syncControls();
    render();
  });

  document.getElementById('chartType').addEventListener('change', function (ev) {
    state.spec = chartPreset(ev.target.value);
    syncControls();
    render();
  });

  document.getElementById('cable').addEventListener('change', function (ev) {
    state.spec.cable = ev.target.value;
    setAllEdges({ cable: ev.target.value });
    syncControls();
    render();
  });

  document.getElementById('motion').addEventListener('change', function (ev) {
    state.spec.motion = ev.target.value;
    setAllEdges({ motion: ev.target.value });
    syncControls();
    render();
  });

  document.getElementById('speed').addEventListener('input', function (ev) {
    state.spec.speed = parseFloat(ev.target.value);
    document.getElementById('speedVal').textContent = state.spec.speed.toFixed(2).replace(/0$/, '') + '×';
    specBox.value = JSON.stringify(state.spec, null, 2);
  });

  document.getElementById('surprise').addEventListener('click', function () {
    var edges = state.spec.edges || [];
    edges.forEach(function (e) {
      e.cable = CABLE_IDS[Math.floor(Math.random() * CABLE_IDS.length)];
      e.motion = MOTION_IDS[Math.floor(Math.random() * MOTION_IDS.length)];
      e.color = COLOR_KEYS[Math.floor(Math.random() * COLOR_KEYS.length)];
    });
    syncControls();
    render();
  });

  document.getElementById('apply').addEventListener('click', function () {
    try {
      state.spec = JSON.parse(specBox.value);
      syncControls();
      render();
    } catch (err) {
      alert('Invalid JSON: ' + err.message);
    }
  });

  document.getElementById('reset').addEventListener('click', function () {
    state.spec = flowPreset();
    syncControls();
    render();
  });

  /* Catalog ------------------------------------------------------------- */

  function miniSpec(cable, motion) {
    return {
      module: 'flow',
      width: 460,
      height: 130,
      cable: cable || 'solid',
      motion: motion || 'none',
      speed: 1,
      columns: [['a'], ['b']],
      nodes: {
        a: { label: 'A', sub: 'input', color: 'blue' },
        b: { label: 'B', sub: 'output', color: 'green' }
      },
      edges: [{ from: 'a', to: 'b' }]
    };
  }

  function makeDemo(host, title, note, spec) {
    var card = document.createElement('div');
    card.className = 'demo';
    var h = document.createElement('h3');
    h.textContent = title;
    var p = document.createElement('p');
    p.textContent = note;
    var mount = document.createElement('div');
    mount.className = 'alur';
    mount.setAttribute('data-module', spec.module);
    mount.setAttribute('data-spec', JSON.stringify(spec));
    card.appendChild(h);
    card.appendChild(p);
    card.appendChild(mount);
    host.appendChild(card);
  }

  var cableHost = document.getElementById('cableCatalog');
  CABLES.forEach(function (c) {
    makeDemo(cableHost, c.label, c.note, miniSpec(c.id, 'none'));
  });

  var motionHost = document.getElementById('motionCatalog');
  MOTIONS.forEach(function (m) {
    makeDemo(motionHost, m.label, m.note, miniSpec('solid', m.id));
  });

  Alur.init(document);
  syncControls();
  render();
})();
